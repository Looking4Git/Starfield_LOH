CREATE VIEW "View_Components_CELL_Helper_Objects" AS
WITH Computed_Refs AS (
    SELECT 
        ref.Component_ID,
        ref.Cell_FormID,
        ref.FormID_Local,
        COALESCE(sys_base.Sig, esm_base.Sig) AS Base_Sig,
        esm_base.FormID_Local AS esm_base_local,
        ref.Sig AS Ref_Sig,
        LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) AS Base_EDID,
        
        -- Deko-Ermittlung
        CASE WHEN 
            COALESCE(sys_base.Sig, esm_base.Sig) IN ('MSTT', 'MISC')
            OR (COALESCE(sys_base.Sig, esm_base.Sig) = 'STAT' AND (
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%clutter%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%deco%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%poster%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%painting%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%rug%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%carpet%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%vase%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%plant%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%pot%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%dish%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%plate%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%cup%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%mug%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%bottle%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%glass%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%tray%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%garbage%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%trash%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%rubble%' OR
                LOWER(COALESCE(sys_base.EDID, esm_base.EDID)) LIKE '%debris%'
            ))
        THEN 1 ELSE 0 END AS is_deco
    FROM Components_ESM_FormIDs ref
    LEFT JOIN Components_System_FormIDs sys_base 
        ON ref.Base_FormID = sys_base.FormID_Local
    LEFT JOIN Components_ESM_FormIDs esm_base 
        ON ref.Base_FormID = esm_base.FormID_Local AND esm_base.Component_ID = ref.Component_ID
    WHERE ref.Sig IN ('REFR', 'ACHR') AND ref.Cell_FormID IS NOT NULL
)
SELECT 
    Component_ID,
    Cell_FormID,
    COUNT(FormID_Local) AS calc_CELL_Refs,
    
    SUM(CASE WHEN Base_Sig = 'CONT' AND esm_base_local IS NOT NULL THEN 1 ELSE 0 END) AS calc_Helper_Objects_Custom_CONT,
    SUM(CASE WHEN Base_Sig = 'TERM' AND esm_base_local IS NOT NULL THEN 1 ELSE 0 END) AS calc_Helper_Objects_Custom_TERM,
    SUM(CASE WHEN Base_Sig = 'BOOK' AND esm_base_local IS NOT NULL THEN 1 ELSE 0 END) AS calc_Helper_Objects_Custom_BOOK,
    SUM(CASE WHEN Base_Sig = 'KEYM' AND esm_base_local IS NOT NULL THEN 1 ELSE 0 END) AS calc_Helper_Objects_Custom_KEYM,
    
    SUM(CASE WHEN Base_Sig = 'FURN' AND esm_base_local IS NOT NULL AND (
        Base_EDID LIKE '%workbench%' OR Base_EDID LIKE '%crafting%' OR Base_EDID LIKE '%cooking%'
    ) THEN 1 ELSE 0 END) AS calc_Helper_Objects_Custom_FURN,
    
    SUM(CASE WHEN Ref_Sig = 'ACHR' AND (
        (esm_base_local IS NOT NULL AND Base_Sig = 'NPC_' AND (
            EXISTS (
                SELECT 1 FROM Components_ESM_References r 
                WHERE r.Component_ID = Computed_Refs.Component_ID AND r.NPC__Dummy > 0
            )
            OR Base_EDID LIKE '%mannequin%' OR Base_EDID LIKE '%dummy%'
        ))
        OR (Base_Sig = 'NPC_' AND (Base_EDID LIKE '%mannequin%' OR Base_EDID LIKE '%dummy%'))
    ) THEN 1 ELSE 0 END) AS calc_Helper_Objects_Custom_ACHR,
    
    (
        SUM(CASE WHEN Base_Sig = 'CONT' AND esm_base_local IS NOT NULL THEN 1 ELSE 0 END) +
        SUM(CASE WHEN Base_Sig = 'TERM' AND esm_base_local IS NOT NULL THEN 1 ELSE 0 END) +
        SUM(CASE WHEN Base_Sig = 'BOOK' AND esm_base_local IS NOT NULL THEN 1 ELSE 0 END) +
        SUM(CASE WHEN Base_Sig = 'KEYM' AND esm_base_local IS NOT NULL THEN 1 ELSE 0 END) +
        SUM(CASE WHEN Base_Sig = 'FURN' AND esm_base_local IS NOT NULL AND (
            Base_EDID LIKE '%workbench%' OR Base_EDID LIKE '%crafting%' OR Base_EDID LIKE '%cooking%'
        ) THEN 1 ELSE 0 END) +
        SUM(CASE WHEN Ref_Sig = 'ACHR' AND (
            (esm_base_local IS NOT NULL AND Base_Sig = 'NPC_' AND (
                EXISTS (
                    SELECT 1 FROM Components_ESM_References r 
                    WHERE r.Component_ID = Computed_Refs.Component_ID AND r.NPC__Dummy > 0
                )
                OR Base_EDID LIKE '%mannequin%' OR Base_EDID LIKE '%dummy%'
            ))
            OR (Base_Sig = 'NPC_' AND (Base_EDID LIKE '%mannequin%' OR Base_EDID LIKE '%dummy%'))
        ) THEN 1 ELSE 0 END)
    ) AS calc_Helper_Objects,
    
    SUM(is_deco) AS calc_Deko_Objects,
    
    SUM(CASE WHEN 
        Base_Sig IN ('DOOR', 'LIGH')
        OR (Base_Sig = 'FURN' AND NOT (
            esm_base_local IS NOT NULL AND (
                Base_EDID LIKE '%workbench%' OR Base_EDID LIKE '%crafting%' OR Base_EDID LIKE '%cooking%'
            )
        ))
        OR (Base_Sig = 'STAT' AND NOT is_deco)
    THEN 1 ELSE 0 END) AS calc_Structural_Refs,
    
    CAST(
        (
            SUM(CASE WHEN Base_Sig = 'CONT' AND esm_base_local IS NOT NULL THEN 1 ELSE 0 END) +
            SUM(CASE WHEN Base_Sig = 'TERM' AND esm_base_local IS NOT NULL THEN 1 ELSE 0 END) +
            SUM(CASE WHEN Base_Sig = 'BOOK' AND esm_base_local IS NOT NULL THEN 1 ELSE 0 END) +
            SUM(CASE WHEN Base_Sig = 'KEYM' AND esm_base_local IS NOT NULL THEN 1 ELSE 0 END) +
            SUM(CASE WHEN Base_Sig = 'FURN' AND esm_base_local IS NOT NULL AND (
                Base_EDID LIKE '%workbench%' OR Base_EDID LIKE '%crafting%' OR Base_EDID LIKE '%cooking%'
            ) THEN 1 ELSE 0 END) +
            SUM(CASE WHEN Ref_Sig = 'ACHR' AND (
                (esm_base_local IS NOT NULL AND Base_Sig = 'NPC_' AND (
                    EXISTS (
                        SELECT 1 FROM Components_ESM_References r 
                        WHERE r.Component_ID = Computed_Refs.Component_ID AND r.NPC__Dummy > 0
                    )
                    OR Base_EDID LIKE '%mannequin%' OR Base_EDID LIKE '%dummy%'
                ))
                OR (Base_Sig = 'NPC_' AND (Base_EDID LIKE '%mannequin%' OR Base_EDID LIKE '%dummy%'))
            ) THEN 1 ELSE 0 END)
        ) AS REAL
    ) / MAX(1, COUNT(FormID_Local) - SUM(is_deco)) AS calc_Helper_Objects_Ratio
    
FROM Computed_Refs
GROUP BY Component_ID, Cell_FormID;

CREATE VIEW "View_Components_ESM_LCTN" AS
SELECT 
    l.ID,
    l.Component_ID,
    l.FormID_Local,
    l.FormID_Keyword,
    COALESCE(ef.EDID, sf.EDID) AS EDID
FROM "Components_ESM_LCTN" l
LEFT JOIN "Components_ESM_FormIDs" ef 
    ON l.FormID_Local = ef.FormID_Local 
   AND l.Component_ID = ef.Component_ID 
   AND ef.Sig = 'LCTN'
LEFT JOIN "Components_System_FormIDs" sf 
    ON l.FormID_Local = sf.FormID_Local 
   AND sf.Sig = 'LCTN';

    COALESCE(ref.MSTT, 0) +

    COALESCE(ref.SFBK, 0) + 
    COALESCE(log10(NULLIF(ref.PCCN, 0)) * 2, 0) + 
    COALESCE(log10(NULLIF(ref.PCBN, 0)) * 2, 0) + 
    COALESCE(log10(NULLIF(ref.LCTN, 0)) * 2, 0) AS dim_calc_outer_POIs,

    MAX(0, COALESCE(ref.NPC__Override, 0) - COALESCE(ref.NPC__KYWD_Companion_Override, 0) - COALESCE(ref.NPC__KYWD_Creature_Override, 0)) AS dim_calc_inner_NPCs,
    MAX(0, COALESCE(ref.NPC_, 0) - COALESCE(ref.NPC__Dummy, 0) - COALESCE(ref.NPC__KYWD_Companion, 0) - COALESCE(ref.NPC__KYWD_Creature, 0)) AS dim_calc_outer_NPCs,
    CASE 
        WHEN (COALESCE(ref.NPC_, 0) > 0 AND COALESCE(ref.NPC_, 0) = COALESCE(ref.NPC__Dummy, 0)) THEN 0 
        ELSE COALESCE(ref.NPC__KYWD_Companion_Override, 0) 
    END AS dim_calc_inner_Begleiter,
    CASE 
        WHEN (COALESCE(ref.NPC_, 0) > 0 AND COALESCE(ref.NPC_, 0) = COALESCE(ref.NPC__Dummy, 0)) THEN 0 
        ELSE COALESCE(ref.NPC__KYWD_Companion, 0) 
    END AS dim_calc_outer_Begleiter,
    COALESCE(ref.NPC__KYWD_Creature_Override, 0) + COALESCE(ref.RACE_Override, 0) + 
    CASE 
        WHEN (COALESCE(ref.RACE_Override, 0) > 0 OR COALESCE(ref.NPC__KYWD_Creature_Override, 0) > 0)
        THEN MAX(0, COALESCE(ref.NPC__Override, 0) - COALESCE(ref.NPC__KYWD_Companion_Override, 0))
        ELSE 0 
    END AS dim_calc_inner_Kreaturen,
    
    CASE 
        WHEN (COALESCE(ref.NPC_, 0) > 0 AND COALESCE(ref.NPC_, 0) = COALESCE(ref.NPC__Dummy, 0) AND COALESCE(ref.NPC__KYWD_Creature, 0) = 0) THEN 0 
        ELSE COALESCE(ref.NPC__KYWD_Creature, 0) + COALESCE(ref.RACE, 0) 
    END + 
    CASE 
        WHEN (COALESCE(ref.RACE, 0) > 0 OR COALESCE(ref.NPC__KYWD_Creature, 0) > 0)
        THEN MAX(0, COALESCE(ref.NPC_, 0) - COALESCE(ref.NPC__Dummy, 0) - COALESCE(ref.NPC__KYWD_Companion, 0))
        ELSE 0 
    END AS dim_calc_outer_Kreaturen,
    CASE 
        WHEN COALESCE(ref.TXST, 0) > 0 THEN COALESCE(ref.TXST_Override, 0)
        ELSE COALESCE((
            SELECT COUNT(*) 
            FROM Components c_ba2 
            JOIN Mods_Components mc_ba2 ON c_ba2.ID = mc_ba2.Component_ID 
            JOIN Mods_Components mc_esm ON mc_ba2.Mod_ID = mc_esm.Mod_ID 
            WHERE mc_esm.Component_ID = ref.Component_ID 
              AND (LOWER(c_ba2.File) LIKE '%texture%.ba2' AND LOWER(c_ba2.File) NOT LIKE '%photomode%')
        ), 0)
    END AS dim_calc_inner_Texturen,
    
    CASE 
        WHEN COALESCE(ref.TXST, 0) > 0 THEN ref.TXST
        ELSE COALESCE((
            SELECT COUNT(*) 
            FROM Components c_ba2 
            JOIN Mods_Components mc_ba2 ON c_ba2.ID = mc_ba2.Component_ID 
            JOIN Mods_Components mc_esm ON mc_ba2.Mod_ID = mc_esm.Mod_ID 
            WHERE mc_esm.Component_ID = ref.Component_ID 
              AND (LOWER(c_ba2.File) LIKE '%texture%.ba2' AND LOWER(c_ba2.File) NOT LIKE '%photomode%')
        ), 0)
    END AS dim_calc_outer_Texturen,
    COALESCE(ref.LIGH_Override, 0) + COALESCE(ref.STDT_Override, 0) AS dim_calc_inner_Licht,
    COALESCE(ref.LIGH, 0) + COALESCE(ref.STDT, 0) AS dim_calc_outer_Licht,
    COALESCE(ref.EFSH_Override, 0) + COALESCE(ref.EFSH_Remove, 0) + 
    COALESCE(ref.IMAD_Override, 0) AS dim_calc_inner_Effekte,
    
    COALESCE(ref.EFSH, 0) + COALESCE(ref.EFSH_Remove, 0) + 
    COALESCE(ref.IMAD, 0) AS dim_calc_outer_Effekte,
    CASE 
        WHEN (COALESCE(ref.SOUN, 0) + COALESCE(ref.MUSC, 0)) > 0 THEN 
            COALESCE(ref.SOUN_Override, 0) + COALESCE(ref.MUSC_Override, 0)
        ELSE COALESCE((
            SELECT COUNT(*) 
            FROM Components c_ba2 
            JOIN Mods_Components mc_ba2 ON c_ba2.ID = mc_ba2.Component_ID 
            JOIN Mods_Components mc_esm ON mc_ba2.Mod_ID = mc_esm.Mod_ID 
            WHERE mc_esm.Component_ID = ref.Component_ID 
              AND (LOWER(c_ba2.File) LIKE '%sound%.ba2' OR LOWER(c_ba2.File) LIKE '%voice%.ba2' OR LOWER(c_ba2.File) LIKE '%audio%.ba2')
        ), 0)
    END AS dim_calc_inner_Sound,
    
    CASE 
        WHEN (COALESCE(ref.SOUN, 0) + COALESCE(ref.MUSC, 0)) > 0 THEN 
            COALESCE(ref.SOUN, 0) + COALESCE(ref.MUSC, 0)
        ELSE COALESCE((

         COALESCE(ref.INFO, 0) + COALESCE(ref.INNR, 0) + COALESCE(ref.KEYM, 0) + 
         COALESCE(ref.KYWD, 0) + COALESCE(ref.LCTN, 0) + COALESCE(ref.LIGH, 0) + 
         COALESCE(ref.LMSW, 0) + COALESCE(ref.LVLI, 0) + COALESCE(ref.LVLN, 0) + 
         COALESCE(ref.LVLP, 0) + COALESCE(ref.MGEF, 0) + COALESCE(ref.MISC, 0) + 
         COALESCE(ref.MSTT, 0) + COALESCE(ref.MUSC, 0) + 
         COALESCE(ref.NAVM, 0) + COALESCE(ref.NPC_, 0) + COALESCE(ref.OMOD, 0) + COALESCE(ref.PERK, 0) + 
         COALESCE(ref.PKIN, 0) + COALESCE(ref.QUST, 0) + COALESCE(ref.RACE, 0) + COALESCE(ref.REGN, 0) + 
         COALESCE(ref.RSPJ, 0) + COALESCE(ref.SFBK, 0) + COALESCE(ref.SOUN, 0) + 
         COALESCE(ref.SPEL, 0) + COALESCE(ref.STAT, 0) + 
         COALESCE(ref.STDT, 0) + COALESCE(ref.TERM, 0) + 
         COALESCE(ref.TXST, 0) + COALESCE(ref.WEAP, 0) + COALESCE(ref.WRLD, 0) + 
         COALESCE(ref.WTHS, 0)), 0
    ) AS calc_Override_Ratio,

    -- Zähler der aktiven funktionalen Bereiche für Overhauls (World, Actors, Effects, Items, Systems)
    (CASE WHEN (COALESCE(ref.CELL_Override, 0) + COALESCE(ref.STAT_Override, 0) + COALESCE(ref.FURN_Override, 0) + 
                COALESCE(ref.WTHS_Override, 0) + COALESCE(ref.WRLD_Override, 0) + COALESCE(ref.RSPJ_Override, 0) + 
                COALESCE(ref.STDT_Override, 0)) > 0 THEN 1 ELSE 0 END) +
    (CASE WHEN (COALESCE(ref.NPC__Override, 0) + COALESCE(ref.PERK_Override, 0) + COALESCE(ref.AVIF_Override, 0)) > 0 THEN 1 ELSE 0 END) +
    (CASE WHEN (COALESCE(ref.SPEL_Override, 0) + COALESCE(ref.MGEF_Override, 0) + COALESCE(ref.ENCH_Override, 0) + COALESCE(ref.ALCH_Override, 0)) > 0 THEN 1 ELSE 0 END) +
    (CASE WHEN (COALESCE(ref.WEAP_Override, 0) + COALESCE(ref.ARMO_Override, 0) + COALESCE(ref.COBJ_Override, 0) + COALESCE(ref.AMMO_Override, 0) + COALESCE(ref.CONT_Override, 0) + COALESCE(ref.MISC_Override, 0)) > 0 THEN 1 ELSE 0 END) +
    (CASE WHEN (COALESCE(ref.GMST_Override, 0) + COALESCE(ref.QUST_Override, 0) + COALESCE(ref.DIAL_Override, 0) + COALESCE(ref.INFO_Override, 0) + COALESCE(ref.TERM_Override, 0) + COALESCE(ref.FLST_Override, 0) + COALESCE(ref.CAMS_Override, 0) + COALESCE(ref.CPTH_Override, 0)) > 0 THEN 1 ELSE 0 END) AS calc_Overhaul_Active_Areas,

    -- Die bisherigen calc_ Spalten
    MAX(0, ref.CELL - ref.CELL_PackIn - COALESCE(ref.CELL_Dummy, 0)) AS calc_CELL_real,
    MAX(0, ref.CELL_Override - ref.CELL_Override_PackIn - COALESCE(ref.CELL_Override_Dummy, 0)) AS calc_CELL_Override_real,
    
    MAX(0, ref.CELL - ref.CELL_PackIn - COALESCE(ref.CELL_Dummy, 0)
        - COALESCE(qry.qry_CELL_City, 0)
        - COALESCE(qry.qry_CELL_Settlement, 0)
        - COALESCE(qry.qry_CELL_Playerhome, 0)
        - COALESCE(qry.qry_CELL_Dungeon, 0)
        - COALESCE(qry.qry_CELL_Ship, 0)
    ) AS calc_CELL_Environment,
    
    MAX(0, ref.CELL_Override - ref.CELL_Override_PackIn - COALESCE(ref.CELL_Override_Dummy, 0)
        - COALESCE(qry.qry_CELL_Override_City, 0)
        - COALESCE(qry.qry_CELL_Override_Settlement, 0)
        - COALESCE(qry.qry_CELL_Override_Playerhome, 0)
        - COALESCE(qry.qry_CELL_Override_Dungeon, 0)
        - COALESCE(qry.qry_CELL_Override_Ship, 0)
    ) AS calc_CELL_Override_Environment,
    
    COALESCE(helper.calc_Helper_Objects, 0) AS calc_Helper_Objects,
    CAST(COALESCE(helper.calc_Helper_Objects, 0) AS REAL) / MAX(1, COALESCE(helper.calc_Helper_Refs_Netto, 0)) AS calc_Helper_Objects_Ratio

FROM Components_ESM_References ref
LEFT JOIN (
    SELECT 
        Component_ID,
        SUM(calc_Helper_Objects) AS calc_Helper_Objects,
        SUM(calc_CELL_Refs - calc_Deko_Objects) AS calc_Helper_Refs_Netto
    FROM View_Components_CELL_Helper_Objects
    GROUP BY Component_ID
) helper ON ref.Component_ID = helper.Component_ID
LEFT JOIN View_Components_References_Qry qry ON ref.Component_ID = qry.Component_ID
-- Join für die BGS-Klassifizierung:
LEFT JOIN Components c ON ref.Component_ID = c.ID
LEFT JOIN Mods_Components mc ON c.ID = mc.Component_ID
LEFT JOIN Mods m ON mc.Mod_ID = m.ID
LEFT JOIN Categories_LOH cat_class ON m.Category_LOH_Classifier = cat_class.ID;

CREATE VIEW View_LoadOrder_Classifier_Results AS
SELECT 
    r.Mod_ID,
    COALESCE(m.Name_Online, m.Name_ContentCatalog) AS Mod_Name,
    r.Component_ID,
    c.File AS Component_File,
    r.Step_ID,
    r.Step_Description,
    cl.Position AS CategoryLOH_Position_Looser,
    cl.Name AS CategoryLOH_Looser,
    cw.Name AS CategoryLOH_Winner,
    crt.Type AS Type,
    r.Reason
FROM LoadOrder_Classifier_Results r
JOIN Mods m ON r.Mod_ID = m.ID
JOIN Components c ON r.Component_ID = c.ID
JOIN Categories_LOH cl ON r.CategoryLOH_ID_Looser = cl.ID
LEFT JOIN Categories_LOH cw ON r.CategoryLOH_ID_Winner = cw.ID
JOIN Classifier_Result_Types crt ON r.Type_ID = crt.ID
ORDER BY r.Mod_ID, r.Component_ID, r.Step_ID, cl.Position;

CREATE VIEW View_Mods_Conflicts_FormID_Base AS
WITH RawPairs AS (
    SELECT DISTINCT
        CASE WHEN mc1.Mod_ID < mc2.Mod_ID THEN mc1.Mod_ID ELSE mc2.Mod_ID END AS Mod_ID_A,
        CASE WHEN mc1.Mod_ID < mc2.Mod_ID THEN mc2.Mod_ID ELSE mc1.Mod_ID END AS Mod_ID_B,
        f1.Master_Component_ID,
        f1.Master_Is_System,
        f1.FormID_Local,
        f1.Sig
    FROM Components_ESM_FormIDs f1
    JOIN Components_ESM_FormIDs f2
      ON f1.Master_Component_ID = f2.Master_Component_ID
     AND f1.FormID_Local = f2.FormID_Local
     AND f1.Master_Is_System = f2.Master_Is_System
     AND f1.Component_ID < f2.Component_ID
    JOIN Mods_Components mc1 ON f1.Component_ID = mc1.Component_ID
    JOIN Mods_Components mc2 ON f2.Component_ID = mc2.Component_ID
    WHERE f1.Is_Override = 1 AND f2.Is_Override = 1 AND f1.Sig NOT IN ('TES4', 'NAVI')
      AND mc1.Mod_ID != mc2.Mod_ID
      AND NOT EXISTS (
          SELECT 1 FROM Components_ESM_Dummies d
          WHERE d.FormID_Local = f1.FormID_Local
            AND (d.Component_ID = f1.Component_ID OR d.Component_ID = f2.Component_ID)
      )
)
SELECT 
    p.Mod_ID_A,
    COALESCE(m1.Name_Online, m1.Name_ContentCatalog, 'Mod ' || p.Mod_ID_A) AS Mod_Name_A,
    p.Mod_ID_B,
    COALESCE(m2.Name_Online, m2.Name_ContentCatalog, 'Mod ' || p.Mod_ID_B) AS Mod_Name_B,
    p.Sig,
    p.FormID_Local AS FormID,
    COALESCE(
        NULLIF(f1.EDID, ''),
        NULLIF(f2.EDID, ''),
        NULLIF(sys.EDID, '')
    ) AS EDID,
    CASE 
        WHEN p.Sig = 'CELL' THEN p.FormID_Local
        ELSE COALESCE(f1.Cell_FormID, f2.Cell_FormID)
    END AS Cell_FormID,
    COALESCE(
        NULLIF(c_mod.EDID, ''),
        NULLIF(c_sys.EDID, '')
    ) AS Cell_EDID,
    COALESCE(f1.Base_FormID, f2.Base_FormID) AS Base_FormID,
    COALESCE(
        NULLIF(b_mod.EDID, ''),
        NULLIF(b_sys.EDID, '')
    ) AS Base_EDID
FROM RawPairs p
JOIN Mods m1 ON p.Mod_ID_A = m1.ID
JOIN Mods m2 ON p.Mod_ID_B = m2.ID
LEFT JOIN Components_ESM_FormIDs f1 
  ON p.FormID_Local = f1.FormID_Local 
 AND f1.Component_ID IN (SELECT Component_ID FROM Mods_Components WHERE Mod_ID = p.Mod_ID_A)
 AND f1.Is_Override = 1
LEFT JOIN Components_ESM_FormIDs f2 
  ON p.FormID_Local = f2.FormID_Local 
 AND f2.Component_ID IN (SELECT Component_ID FROM Mods_Components WHERE Mod_ID = p.Mod_ID_B)
 AND f2.Is_Override = 1
LEFT JOIN Components_System_FormIDs sys 
  ON p.FormID_Local = sys.FormID_Local AND p.Sig = sys.Sig
LEFT JOIN Components_ESM_FormIDs c_mod
  ON c_mod.FormID_Local = (CASE WHEN p.Sig = 'CELL' THEN p.FormID_Local ELSE COALESCE(f1.Cell_FormID, f2.Cell_FormID) END)
 AND c_mod.Sig = 'CELL'
LEFT JOIN Components_System_FormIDs c_sys
  ON c_sys.FormID_Local = (CASE WHEN p.Sig = 'CELL' THEN p.FormID_Local ELSE COALESCE(f1.Cell_FormID, f2.Cell_FormID) END)
 AND c_sys.Sig = 'CELL'
LEFT JOIN Components_ESM_FormIDs b_mod
  ON b_mod.FormID_Local = COALESCE(f1.Base_FormID, f2.Base_FormID)
LEFT JOIN Components_System_FormIDs b_sys
  ON b_sys.FormID_Local = COALESCE(f1.Base_FormID, f2.Base_FormID)
GROUP BY p.Mod_ID_A, p.Mod_ID_B, p.FormID_Local, p.Sig;
